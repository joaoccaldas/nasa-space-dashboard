(() => {
  'use strict';
  const THREE_URL = 'https://cdn.jsdelivr.net/npm/three@0.184.0/build/three.module.js';
  const reduceMotion = matchMedia('(prefers-reduced-motion: reduce)').matches;
  const scenes = [];
  let THREE;

  function visible(element) {
    if (!element || document.hidden) return false;
    const rect = element.getBoundingClientRect();
    return rect.bottom > -120 && rect.top < innerHeight + 120;
  }

  function makeRenderer(host) {
    const renderer = new THREE.WebGLRenderer({ alpha: true, antialias: true, powerPreference: 'high-performance' });
    renderer.setPixelRatio(Math.min(devicePixelRatio || 1, innerWidth < 700 ? 1.35 : 1.7));
    renderer.setClearColor(0x000000, 0);
    renderer.outputColorSpace = THREE.SRGBColorSpace;
    renderer.toneMapping = THREE.ACESFilmicToneMapping;
    renderer.toneMappingExposure = 1.15;
    renderer.domElement.className = 'three-stage';
    renderer.domElement.setAttribute('aria-hidden', 'true');
    host.prepend(renderer.domElement);
    return renderer;
  }

  function resizeStage(stage) {
    const rect = stage.host.getBoundingClientRect();
    const width = Math.max(1, rect.width), height = Math.max(1, rect.height);
    stage.renderer.setSize(width, height, false);
    stage.camera.aspect = width / height;
    stage.camera.updateProjectionMatrix();
  }

  function addStars(scene, count = 700, radius = 18) {
    const positions = new Float32Array(count * 3);
    const colors = new Float32Array(count * 3);
    const palette = [new THREE.Color('#ffffff'), new THREE.Color('#8cf4ff'), new THREE.Color('#c6a6ff')];
    for (let i = 0; i < count; i++) {
      const r = radius * (.45 + Math.random() * .55);
      const theta = Math.random() * Math.PI * 2;
      const phi = Math.acos(2 * Math.random() - 1);
      positions[i * 3] = r * Math.sin(phi) * Math.cos(theta);
      positions[i * 3 + 1] = r * Math.cos(phi);
      positions[i * 3 + 2] = r * Math.sin(phi) * Math.sin(theta);
      const c = palette[i % palette.length]; colors[i * 3] = c.r; colors[i * 3 + 1] = c.g; colors[i * 3 + 2] = c.b;
    }
    const geometry = new THREE.BufferGeometry();
    geometry.setAttribute('position', new THREE.BufferAttribute(positions, 3));
    geometry.setAttribute('color', new THREE.BufferAttribute(colors, 3));
    const points = new THREE.Points(geometry, new THREE.PointsMaterial({ size: .035, transparent: true, opacity: .72, vertexColors: true, depthWrite: false }));
    scene.add(points);
    return points;
  }

  function atmosphere(radius, color) {
    const material = new THREE.ShaderMaterial({
      side: THREE.BackSide,
      transparent: true,
      blending: THREE.AdditiveBlending,
      depthWrite: false,
      uniforms: { glowColor: { value: new THREE.Color(color) } },
      vertexShader: 'varying vec3 vNormal; void main(){vNormal=normalize(normalMatrix*normal);gl_Position=projectionMatrix*modelViewMatrix*vec4(position,1.0);}',
      fragmentShader: 'uniform vec3 glowColor; varying vec3 vNormal; void main(){float i=pow(0.72-dot(vNormal,vec3(0.0,0.0,1.0)),2.4);gl_FragColor=vec4(glowColor,i*.72);}'
    });
    return new THREE.Mesh(new THREE.SphereGeometry(radius, 64, 64), material);
  }

  function mountHero() {
    const host = document.querySelector('.hero-art');
    if (!host || host.dataset.threeMounted) return;
    host.dataset.threeMounted = '1';
    const renderer = makeRenderer(host);
    const scene = new THREE.Scene();
    const camera = new THREE.PerspectiveCamera(38, 1, .1, 100);
    camera.position.set(0, .25, 7.2);
    const group = new THREE.Group(); scene.add(group);
    addStars(scene, innerWidth < 700 ? 360 : 760, 19);

    const planet = new THREE.Mesh(
      new THREE.SphereGeometry(1.62, innerWidth < 700 ? 48 : 72, innerWidth < 700 ? 48 : 72),
      new THREE.MeshPhysicalMaterial({ color: 0x4d5bff, roughness: .56, metalness: .08, clearcoat: .3, clearcoatRoughness: .45, emissive: 0x09164a, emissiveIntensity: .55 })
    );
    planet.rotation.z = -.18; group.add(planet);
    const atmosphereMesh = atmosphere(1.83, '#72e8ff'); group.add(atmosphereMesh);

    const ringMaterial = new THREE.MeshBasicMaterial({ color: 0x8cf4ff, transparent: true, opacity: .48, side: THREE.DoubleSide, blending: THREE.AdditiveBlending, depthWrite: false });
    const ring = new THREE.Mesh(new THREE.RingGeometry(2.05, 2.12, 160), ringMaterial); ring.rotation.x = 1.27; ring.rotation.z = -.32; group.add(ring);
    const ring2 = new THREE.Mesh(new THREE.RingGeometry(2.58, 2.61, 160), ringMaterial.clone()); ring2.material.color.set('#ff7cb8'); ring2.material.opacity = .24; ring2.rotation.x = 1.05; ring2.rotation.z = .27; group.add(ring2);

    const moon = new THREE.Mesh(new THREE.SphereGeometry(.18, 28, 28), new THREE.MeshStandardMaterial({ color: 0xffd67a, emissive: 0x5a3510, emissiveIntensity: .8, roughness: .48 })); group.add(moon);
    const satellite = new THREE.Group();
    const body = new THREE.Mesh(new THREE.BoxGeometry(.18,.12,.28), new THREE.MeshStandardMaterial({color:0xe8f7ff,metalness:.55,roughness:.3})); satellite.add(body);
    const panelMat = new THREE.MeshStandardMaterial({color:0x1849a9,emissive:0x06132e,emissiveIntensity:.7});
    [-1,1].forEach(side=>{const panel=new THREE.Mesh(new THREE.BoxGeometry(.42,.018,.16),panelMat);panel.position.x=side*.32;satellite.add(panel);}); group.add(satellite);

    scene.add(new THREE.AmbientLight(0x8da8ff, 1.5));
    const key = new THREE.DirectionalLight(0xffffff, 5.5); key.position.set(-4, 3, 6); scene.add(key);
    const rim = new THREE.PointLight(0xff5fa7, 22, 12, 2); rim.position.set(4, -1, 3); scene.add(rim);
    const cyan = new THREE.PointLight(0x72e8ff, 18, 10, 2); cyan.position.set(-3, -2, 3); scene.add(cyan);

    const stage = { host, renderer, scene, camera, group, planet, atmosphereMesh, ring, ring2, moon, satellite, pointer: new THREE.Vector2(), type: 'hero' };
    scenes.push(stage); resizeStage(stage);
    new ResizeObserver(() => resizeStage(stage)).observe(host);
    host.addEventListener('pointermove', event => { const rect = host.getBoundingClientRect(); stage.pointer.set((event.clientX-rect.left)/rect.width-.5,(event.clientY-rect.top)/rect.height-.5); }, { passive: true });
    host.addEventListener('pointerleave', () => stage.pointer.set(0,0));
    host.classList.add('three-ready');
    const label = document.createElement('span'); label.className='three-label'; label.textContent='Interactive orbital model'; host.append(label);
  }

  function mountAtlas() {
    const host = document.querySelector('.atlas-visual');
    if (!host || host.dataset.threeMounted) return;
    host.dataset.threeMounted='1';
    const renderer = makeRenderer(host);
    const scene = new THREE.Scene();
    const camera = new THREE.PerspectiveCamera(43,1,.1,120); camera.position.set(0,.25,7.8);
    const group = new THREE.Group(); scene.add(group);
    addStars(scene, innerWidth < 700 ? 240 : 520, 22);

    const nucleus = new THREE.Mesh(new THREE.IcosahedronGeometry(.38,4),new THREE.MeshPhysicalMaterial({color:0xffd4a1,emissive:0xff5f9f,emissiveIntensity:1.45,roughness:.72,clearcoat:.25})); group.add(nucleus);
    const corona = atmosphere(.62,'#ff7cb8'); group.add(corona);
    const tailCount = innerWidth < 700 ? 520 : 1100;
    const positions = new Float32Array(tailCount*3); const colors = new Float32Array(tailCount*3);
    const c1 = new THREE.Color('#dffcff'), c2 = new THREE.Color('#72e8ff');
    for(let i=0;i<tailCount;i++){const t=Math.random();const spread=.06+t*.32;positions[i*3]=-t*5.2+Math.random()*.18;positions[i*3+1]=(Math.random()-.5)*spread;positions[i*3+2]=(Math.random()-.5)*spread;const c=c1.clone().lerp(c2,t);colors[i*3]=c.r;colors[i*3+1]=c.g;colors[i*3+2]=c.b;}
    const tailGeo=new THREE.BufferGeometry();tailGeo.setAttribute('position',new THREE.BufferAttribute(positions,3));tailGeo.setAttribute('color',new THREE.BufferAttribute(colors,3));
    const tail=new THREE.Points(tailGeo,new THREE.PointsMaterial({size:.035,transparent:true,opacity:.62,vertexColors:true,blending:THREE.AdditiveBlending,depthWrite:false})); group.add(tail);

    const orbitMaterial=new THREE.LineBasicMaterial({color:0x8b7cff,transparent:true,opacity:.48});
    [2.7,3.55].forEach((rx,index)=>{const points=[];for(let i=0;i<=180;i++){const a=i/180*Math.PI*2;points.push(new THREE.Vector3(Math.cos(a)*rx,Math.sin(a)*(1.05+index*.25),-.35-index*.3));}const geo=new THREE.BufferGeometry().setFromPoints(points);const line=new THREE.Line(geo,orbitMaterial.clone());line.rotation.z=index?.28:-.18;group.add(line);});
    const radar=new THREE.Mesh(new THREE.RingGeometry(1.15,1.17,96),new THREE.MeshBasicMaterial({color:0x72e8ff,transparent:true,opacity:.22,side:THREE.DoubleSide}));radar.rotation.x=Math.PI/2;group.add(radar);

    scene.add(new THREE.AmbientLight(0x6d83c6,1.35));const light=new THREE.PointLight(0xffffff,34,12,2);light.position.set(1.8,2.2,4);scene.add(light);const pink=new THREE.PointLight(0xff6ea8,18,9,2);pink.position.set(-2,-1,2);scene.add(pink);
    const stage={host,renderer,scene,camera,group,nucleus,corona,tail,radar,pointer:new THREE.Vector2(),type:'atlas'};scenes.push(stage);resizeStage(stage);new ResizeObserver(()=>resizeStage(stage)).observe(host);
    host.addEventListener('pointermove',event=>{const rect=host.getBoundingClientRect();stage.pointer.set((event.clientX-rect.left)/rect.width-.5,(event.clientY-rect.top)/rect.height-.5);},{passive:true});host.addEventListener('pointerleave',()=>stage.pointer.set(0,0));
    const label=document.createElement('span');label.className='three-label';label.textContent='Interactive 3D model · not to scale';host.append(label);
  }

  function animate(time=0){
    scenes.forEach(stage=>{
      if(!visible(stage.host)) return;
      const t=time*.001;
      if(stage.type==='hero'){
        stage.planet.rotation.y=t*.09; stage.atmosphereMesh.rotation.y=-t*.04; stage.ring.rotation.z=-.32+Math.sin(t*.25)*.025; stage.ring2.rotation.z=.27-Math.sin(t*.21)*.035;
        stage.moon.position.set(Math.cos(t*.55)*2.55,Math.sin(t*.55)*.8,Math.sin(t*.55)*1.2);stage.satellite.position.set(Math.cos(-t*.38)*3.05,Math.sin(-t*.38)*1.1,Math.sin(-t*.38)*1.4);stage.satellite.rotation.y=t;
        stage.group.rotation.y+=(stage.pointer.x*.22-stage.group.rotation.y)*.035;stage.group.rotation.x+=(-stage.pointer.y*.16-stage.group.rotation.x)*.035;
      }else{
        stage.group.rotation.y+=(stage.pointer.x*.18-stage.group.rotation.y)*.03;stage.group.rotation.x+=(-stage.pointer.y*.12-stage.group.rotation.x)*.03;stage.nucleus.rotation.x=t*.18;stage.nucleus.rotation.y=t*.24;stage.corona.scale.setScalar(1+Math.sin(t*2)*.025);stage.radar.rotation.z=t*.35;
      }
      stage.renderer.render(stage.scene,stage.camera);
    });
    if(!reduceMotion) requestAnimationFrame(animate);
  }

  async function init(){
    try{THREE=await import(THREE_URL);mountHero();mountAtlas();animate(performance.now());if(reduceMotion) scenes.forEach(stage=>stage.renderer.render(stage.scene,stage.camera));}
    catch(error){document.documentElement.classList.add('three-unavailable');console.warn('Three.js enhancement unavailable',error);}
  }
  init();
})();

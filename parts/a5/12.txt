(() => {
  'use strict';
  const reduceMotion = matchMedia('(prefers-reduced-motion: reduce)').matches;
  const $ = (selector, root = document) => root.querySelector(selector);
  const $$ = (selector, root = document) => [...root.querySelectorAll(selector)];
  const colors = ['#72e8ff', '#8b7cff', '#ff7cb8', '#ffd67a', '#71e5b3'];

  function mountGlobalLayers() {
    if (!$('#cosmicField')) {
      const canvas = document.createElement('canvas');
      canvas.id = 'cosmicField'; canvas.setAttribute('aria-hidden', 'true');
      document.body.prepend(canvas);
    }
    if (!$('.route-warp')) {
      const warp = document.createElement('div'); warp.className = 'route-warp'; warp.setAttribute('aria-hidden', 'true'); document.body.append(warp);
    }
    if (!$('.scroll-progress')) {
      const progress = document.createElement('div'); progress.className = 'scroll-progress'; progress.setAttribute('aria-hidden', 'true'); document.body.append(progress);
    }
  }

  function routeName() {
    const name = location.hash.replace('#', '');
    return ['home', 'discover', 'learn', 'observe', 'play'].includes(name) ? name : 'home';
  }

  function applyRouteIdentity() {
    document.body.dataset.route = routeName();
    const positions = { home: ['78%', '18%'], discover: ['88%', '22%'], learn: ['14%', '28%'], observe: ['52%', '12%'], play: ['82%', '18%'] };
    const [x, y] = positions[document.body.dataset.route];
    document.body.style.setProperty('--route-x', x); document.body.style.setProperty('--route-y', y);
  }

  function initRouteWarp() {
    addEventListener('hashchange', applyRouteIdentity);
    document.addEventListener('pointerdown', event => {
      const target = event.target.closest('[data-route]');
      if (!target || target.dataset.route === routeName() || reduceMotion) return;
      document.body.style.setProperty('--warp-x', `${event.clientX || innerWidth / 2}px`);
      document.body.style.setProperty('--warp-y', `${event.clientY || innerHeight / 2}px`);
      document.body.classList.remove('route-warping');
      void document.body.offsetWidth;
      document.body.classList.add('route-warping');
      setTimeout(() => document.body.classList.remove('route-warping'), 560);
    }, { capture: true });
    applyRouteIdentity();
  }

  function initCosmicField() {
    const canvas = $('#cosmicField'); if (!canvas) return;
    const ctx = canvas.getContext('2d');
    let width = 0, height = 0, dpr = 1, stars = [], comet = 0;
    function resize() {
      width = innerWidth; height = innerHeight; dpr = Math.min(devicePixelRatio || 1, 2);
      canvas.width = Math.round(width * dpr); canvas.height = Math.round(height * dpr);
      canvas.style.width = `${width}px`; canvas.style.height = `${height}px`; ctx.setTransform(dpr, 0, 0, dpr, 0, 0);
      const count = Math.max(45, Math.min(130, Math.round((width * height) / 11000)));
      stars = Array.from({ length: count }, (_, i) => ({ x: Math.random() * width, y: Math.random() * height, r: .35 + Math.random() * 1.4, a: .18 + Math.random() * .72, speed: .03 + Math.random() * .12, phase: Math.random() * Math.PI * 2, tint: i % 9 === 0 ? colors[i % colors.length] : '#ffffff' }));
      draw(performance.now());
    }
    function draw(time) {
      ctx.clearRect(0, 0, width, height);
      stars.forEach(star => {
        const twinkle = reduceMotion ? star.a : star.a * (.72 + Math.sin(time * .0012 + star.phase) * .28);
        ctx.globalAlpha = Math.max(.08, twinkle); ctx.fillStyle = star.tint;
        ctx.beginPath(); ctx.arc(star.x, star.y, star.r, 0, Math.PI * 2); ctx.fill();
        if (!reduceMotion) { star.y += star.speed; if (star.y > height + 3) { star.y = -3; star.x = Math.random() * width; } }
      });
      if (!reduceMotion) {
        comet = (comet + .0018) % 1.45;
        const x = -120 + comet * (width + 260), y = height * .18 + Math.sin(comet * 5) * 80;
        const gradient = ctx.createLinearGradient(x - 110, y + 40, x, y);
        gradient.addColorStop(0, 'transparent'); gradient.addColorStop(1, 'rgba(114,232,255,.7)');
        ctx.globalAlpha = .42; ctx.strokeStyle = gradient; ctx.lineWidth = 2; ctx.beginPath(); ctx.moveTo(x - 110, y + 40); ctx.lineTo(x, y); ctx.stroke();
      }
      ctx.globalAlpha = 1;
      if (!reduceMotion) requestAnimationFrame(draw);
    }
    addEventListener('resize', resize, { passive: true }); resize();
  }

  function initHeroUniverse() {
    const hero = $('.hero-art'); if (!hero || $('.hero-orbit-canvas', hero)) return;
    const canvas = document.createElement('canvas'); canvas.className = 'hero-orbit-canvas'; canvas.setAttribute('aria-hidden', 'true'); hero.append(canvas);
    ['one', 'two'].forEach(name => { const beacon = document.createElement('i'); beacon.className = `hero-beacon ${name}`; beacon.setAttribute('aria-hidden', 'true'); hero.append(beacon); });
    const ctx = canvas.getContext('2d'); let w = 0, h = 0, dpr = 1;
    function resize() { const rect = hero.getBoundingClientRect(); w = rect.width; h = rect.height; dpr = Math.min(devicePixelRatio || 1, 2); canvas.width = Math.max(1, Math.round(w * dpr)); canvas.height = Math.max(1, Math.round(h * dpr)); canvas.style.width = `${w}px`; canvas.style.height = `${h}px`; ctx.setTransform(dpr, 0, 0, dpr, 0, 0); }
    function render(time) {
      ctx.clearRect(0, 0, w, h); const cx = w * .5, cy = h * .52;
      const paths = [{ rx: w * .37, ry: h * .16, speed: .00026, color: '#72e8ff' }, { rx: w * .29, ry: h * .26, speed: -.00019, color: '#ff7cb8' }, { rx: w * .43, ry: h * .32, speed: .00012, color: '#ffd67a' }];
      paths.forEach((path, index) => {
        ctx.save(); ctx.translate(cx, cy); ctx.rotate(-.24 + index * .33); ctx.strokeStyle = `${path.color}44`; ctx.lineWidth = 1.4; ctx.beginPath(); ctx.ellipse(0, 0, path.rx, path.ry, 0, 0, Math.PI * 2); ctx.stroke();
        const angle = reduceMotion ? index * 2.1 : time * path.speed + index * 2.1;
        const x = Math.cos(angle) * path.rx, y = Math.sin(angle) * path.ry;
        ctx.shadowColor = path.color; ctx.shadowBlur = 14; ctx.fillStyle = path.color; ctx.beginPath(); ctx.arc(x, y, 3.4 + index, 0, Math.PI * 2); ctx.fill(); ctx.restore();
      });
      if (!reduceMotion) requestAnimationFrame(render);
    }
    const observer = new ResizeObserver(resize); observer.observe(hero); resize(); render(performance.now());
    if (matchMedia('(hover:hover)').matches && !reduceMotion) {
      hero.addEventListener('pointermove', event => { const rect = hero.getBoundingClientRect(); const px = (event.clientX - rect.left) / rect.width - .5; const py = (event.clientY - rect.top) / rect.height - .5; hero.style.setProperty('--tilt-y', `${px * 7}deg`); hero.style.setProperty('--tilt-x', `${py * -6}deg`); });
      hero.addEventListener('pointerleave', () => { hero.style.setProperty('--tilt-y', '0deg'); hero.style.setProperty('--tilt-x', '0deg'); });
    }
  }
